\exercise[3 points]
Show that $\sum_{k=1}^{n} k = \frac{n(n+1)}{2}$ for every $n \geq 1$.

\begin{solution}
By induction on $n$. For $n = 1$ both sides are $1$. If it holds for $n$, then
\[
  \sum_{k=1}^{n+1} k = \frac{n(n+1)}{2} + (n+1) = \frac{(n+1)(n+2)}{2}. \qedhere
\]
\end{solution}
